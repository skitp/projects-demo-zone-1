WITH LatestChanges AS (
    SELECT
        /* business columns including candidate keys */,
        __$operation, __$start_lsn, __$command_id, __$seqval,
        ROW_NUMBER() OVER (
            PARTITION BY code, loctypenum, uselocnum   -- your candidate keys
            ORDER BY
                __$start_lsn DESC,
                __$command_id DESC,
                __$seqval DESC,
                __$operation DESC          -- 4 > 3 so after-image wins
        ) AS rn
    FROM cdc.dbo_wclocnam_CT
    WHERE __$operation IN (1, 2, 3, 4)
      AND __$start_lsn >= @from_lsn
)
SELECT
    /* business columns */,
    __$operation  AS __operation,
    __$start_lsn  AS __start_lsn,
    __$command_id AS __command_id,
    __$seqval     AS __seqval
FROM LatestChanges
WHERE rn = 1;

# Prefer higher __operation on ties (4 > 3, 2 > 1)
order_exprs = [col(c).desc() for c in order_cols] + [col("__operation").desc()]

w = Window.partitionBy(*self._candidate_keys).orderBy(*order_exprs)
latest = (
    source_data.withColumn("_rn", row_number().over(w))
    .filter(col("_rn") == 1)
    .drop("_rn")
)

to_delete = latest.filter(col("__operation").isin(1, 3))
to_upsert = latest.filter(col("__operation").isin(2, 4))
